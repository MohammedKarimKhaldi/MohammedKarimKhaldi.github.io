Overview

This project demonstrates the use of Monte Carlo methods for option pricing and risk analysis, implemented in C++ for high performance and scalability.

Sample C++ Code: European Call Option Pricing

// Monte Carlo simulation for European Call Option Pricing (Black-Scholes)
#include <iostream>
#include <random>
#include <cmath>

int main() {
    // Option parameters
    double S0 = 100.0;   // Initial stock price
    double K = 100.0;    // Strike price
    double r = 0.05;     // Risk-free rate
    double sigma = 0.2;  // Volatility
    double T = 1.0;      // Time to maturity (years)
    int N = 100000;      // Number of simulations

    std::mt19937 gen(42); // Random number generator with fixed seed
    std::normal_distribution<> dist(0.0, 1.0);

    double payoff_sum = 0.0;
    for (int i = 0; i < N; ++i) {
        double Z = dist(gen);
        double ST = S0 * std::exp((r - 0.5 * sigma * sigma) * T + sigma * std::sqrt(T) * Z);
        double payoff = std::max(ST - K, 0.0);
        payoff_sum += payoff;
    }
    double price = std::exp(-r * T) * (payoff_sum / N);
    std::cout << "Estimated European Call Option Price: " << price << std::endl;
    return 0;
}

How to Compile and Run

// Save as monte_carlo_option.cpp
g++ -O2 -std=c++11 monte_carlo_option.cpp -o monte_carlo_option
./monte_carlo_option

Estimated European Call Option Price: 10.43
